{{- /* A responsive picture. Pass a dict: img the picture (required) alt description sizes how wide the picture is shown, e.g. "(min-width: 60rem) 50vw, 100vw" widths widths to make, e.g. (slice 480 960 1600) class class for the eager true for a picture at the very top of a page crop optional shape to crop to, as "WIDTHxHEIGHT", e.g. "3x2" absolute true to write full addresses (for the feed) Pictures are resized and converted to the formats listed in params.imageFormats. Animated GIFs and SVGs are passed through untouched. */ -}} {{- $img := .img -}} {{- $alt := .alt | default "" -}} {{- $sizes := .sizes | default "100vw" -}} {{- $widths := .widths | default (slice 480 800 1200 1600 2400) -}} {{- $class := .class -}} {{- $absolute := .absolute | default false -}} {{- $loading := cond (eq .eager true) "eager" "lazy" -}} {{- $sub := $img.MediaType.SubType -}} {{- if or (eq $sub "gif") (eq $sub "svg") (ne $img.ResourceType "image") -}} {{- else -}} {{- $hint := cond (eq $sub "png") "drawing" "photo" -}} {{- $formats := site.Params.imageFormats | default (slice "webp") -}} {{- $base := partial "image-base.html" $img -}} {{- /* Never enlarge: keep the widths smaller than the picture, and add its own width. */ -}} {{- $use := slice -}} {{- range $widths }}{{ if lt . $base.Width }}{{ $use = $use | append . }}{{ end }}{{ end -}} {{- $largest := index $widths (sub (len $widths) 1) -}} {{- if le $base.Width $largest }}{{ $use = $use | append $base.Width }}{{ end -}} {{- $ratio := div (float $base.Height) (float $base.Width) -}} {{- with .crop -}} {{- $parts := split . "x" -}} {{- $ratio = div (float (index $parts 1)) (float (index $parts 0)) -}} {{- end -}} {{- $crop := .crop -}} {{- range $i, $format := $formats }} {{- $set := slice -}} {{- $last := false -}} {{- range $use }} {{- $out := false -}} {{- if $crop }} {{- $out = $base.Fill (printf "%dx%d %s %s" . (int (math.Round (mul (float .) $ratio))) $format $hint) -}} {{- else }} {{- $out = $base.Resize (printf "%dx %s %s" . $format $hint) -}} {{- end }} {{- $set = $set | append (printf "%s %dw" (cond $absolute $out.Permalink $out.RelPermalink) $out.Width) -}} {{- $last = $out -}} {{- end }} {{- if eq (add $i 1) (len $formats) }} {{- /* The last format in the list is what a browser gets when it understands none of the others. */}} {{- else }} {{- end }} {{- end }} {{- end -}}
