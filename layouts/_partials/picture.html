{{- /* A responsive picture. Pass a dict: img the picture (required) alt description sizes how wide the picture is shown, e.g. "(min-width: 60rem) 50vw, 100vw" widths widths to make, e.g. (slice 480 960 1600) class class for the eager true for a picture at the very top of a page crop optional "WIDTHxHEIGHT" ratio to crop to, e.g. "3x2" Pictures are resized and converted to the formats listed in params.imageFormats. Animated GIFs and SVGs are passed through untouched. */ -}} {{- $img := .img -}} {{- $alt := .alt | default "" -}} {{- $sizes := .sizes | default "100vw" -}} {{- $widths := .widths | default (slice 480 800 1200 1600 2400) -}} {{- $class := .class -}} {{- $loading := cond (eq .eager true) "eager" "lazy" -}} {{- $sub := $img.MediaType.SubType -}} {{- if or (eq $sub "gif") (eq $sub "svg") (ne $img.ResourceType "image") -}} {{- else -}} {{- $hint := cond (eq $sub "png") "drawing" "photo" -}} {{- $formats := site.Params.imageFormats | default (slice "webp") -}} {{- /* Never enlarge: keep the widths smaller than the original, and add the original width itself. */ -}} {{- $use := slice -}} {{- range $widths }}{{ if lt . $img.Width }}{{ $use = $use | append . }}{{ end }}{{ end -}} {{- $largest := index $widths (sub (len $widths) 1) -}} {{- if le $img.Width $largest }}{{ $use = $use | append $img.Width }}{{ end -}} {{- $ratio := div (float $img.Height) (float $img.Width) -}} {{- with .crop -}} {{- $parts := split . "x" -}} {{- $ratio = div (float (index $parts 1)) (float (index $parts 0)) -}} {{- end -}} {{- $crop := .crop -}} {{- $fallback := false -}} {{- range $i, $format := $formats }} {{- $set := slice -}} {{- $last := false -}} {{- range $use }} {{- $out := false -}} {{- if $crop }} {{- $out = $img.Fill (printf "%dx%d %s %s" . (int (math.Round (mul (float .) $ratio))) $format $hint) -}} {{- else }} {{- $out = $img.Resize (printf "%dx %s %s" . $format $hint) -}} {{- end }} {{- $set = $set | append (printf "%s %dw" $out.RelPermalink $out.Width) -}} {{- $last = $out -}} {{- end }} {{- if eq (add $i 1) (len $formats) }} {{- /* The last format is the one every browser gets if it understands none of the others. */}} {{- else }} {{- end }} {{- end }} {{- end -}}
