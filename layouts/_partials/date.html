{{- /* Shows a page's date. Pass (dict "page" . "format" "short|month|year|long"). A `period` in the front matter ("March – June 2026") replaces the date where a month is asked for. */ -}} {{- $page := .page -}} {{- $format := .format | default "month" -}} {{- $layout := "January 2006" -}} {{- if eq $format "year" }}{{ $layout = "2006" }}{{ else if eq $format "long" }}{{ $layout = "2 January 2006" }}{{ else if eq $format "short" }}{{ $layout = "2 Jan 2006" }}{{ end -}} {{- if not $page.Date.IsZero -}} {{- if and $page.Params.period (eq $format "month") }}{{ $page.Params.period }}{{ else }}{{ $page.Date | time.Format $layout }}{{ end -}} {{- end -}}
