{{- /* Returns a drawing as inline SVG, or nothing. Pass (dict "page" . "name" "figure.svg" "class" "…"). The file is looked for beside the page first, then in assets/figures. A drawing may hold shapes and text only. A file with a script, a link, an embedded picture or a reference to anything outside itself stops the build. */ -}} {{- $page := .page -}} {{- $name := .name -}} {{- $class := .class | default "" -}} {{- $svg := "" -}} {{- with $name -}} {{- $file := false -}} {{- with $page }}{{ with .Resources.Get $name }}{{ $file = . }}{{ end }}{{ end -}} {{- if not $file }}{{ with resources.Get (printf "figures/%s" $name) }}{{ $file = . }}{{ end }}{{ end -}} {{- if $file -}} {{- $svg = $file.Content -}} {{- $svg = replaceRE `(?s)<\?xml.*?\?>` "" $svg -}} {{- $svg = replaceRE `(?s)` "" $svg -}} {{- $svg = replaceRE `(?s)` "" $svg -}} {{- $svg = replaceRE `(?s)]*>.*?` "" $svg -}} {{- $svg = trim $svg " \n\r\t" -}} {{- $bad := findRE `(?i)<\s*(script|foreignObject|image|use|iframe|a|animate\w*|set|style|link)\b|<[^>]*\s(on[a-z]+|href|xlink:href)\s*=|<[^>]*url\s*\(|javascript:|." $name $page.Path -}} {{- $svg = "" -}} {{- else -}} {{- $svg = replaceRE `^]*?)\s+class="[^"]*"` "]*?)\s+xmlns="[^"]*"` "
